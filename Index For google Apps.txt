  <body>
    <h2>Store, Edit & Upload HTML/ZIP</h2>

    <div class="file-section">
      <label for="fileInput"><b>Load File (.html or .zip):</b></label><br><br>
      <input type="file" id="fileInput" accept=".html,.htm,.zip" onchange="handleFileUpload(event)">
    </div>

    <label for="editor"><b>HTML Editor:</b></label><br>
    <textarea id="editor"></textarea><br>

    <div class="btn-group">
      <button onclick="saveData()">Save Changes</button>
      <button onclick="openInAboutBlank()">Open in New Window (about:blank)</button>
      <button onclick="loadIntoIframe()">Load into Frame (about:blank)</button>
      <span id="status"></span>
    </div>

    <h3>Inline Preview</h3>
    <div id="preview">Loading...</div>

    <h3>about:blank Frame Preview</h3>
    <iframe id="blankFrame" src="about:blank"></iframe>

    <script>
      window.onload = function() {
        google.script.run.withSuccessHandler(function(content) {
          document.getElementById('editor').value = content;
          document.getElementById('preview').innerHTML = content;
        }).getHtmlContent();
      };

      function saveData() {
        var updatedHtml = document.getElementById('editor').value;
        document.getElementById('status').innerText = 'Saving...';
        
        google.script.run.withSuccessHandler(function(response) {
          document.getElementById('status').innerText = response;
          document.getElementById('preview').innerHTML = updatedHtml;
        }).saveHtmlContent(updatedHtml);
      }

      function openInAboutBlank() {
        var htmlContent = document.getElementById('editor').value;
        var newWin = window.open('about:blank', '_blank');
        if (newWin) {
          newWin.document.open();
          newWin.document.write(htmlContent);
          newWin.document.close();
        } else {
          alert('Popup blocked! Please allow popups for this web app.');
        }
      }

      function loadIntoIframe() {
        var htmlContent = document.getElementById('editor').value;
        var frame = document.getElementById('blankFrame');
        var frameDoc = frame.contentDocument || frame.contentWindow.document;
        frameDoc.open();
        frameDoc.write(htmlContent);
        frameDoc.close();
      }

      // Handle loading both .html and .zip files
      function handleFileUpload(event) {
        var file = event.target.files[0];
        if (!file) return;

        var fileName = file.name.toLowerCase();

        if (fileName.endsWith('.html') || fileName.endsWith('.htm')) {
          var reader = new FileReader();
          reader.onload = function(e) {
            document.getElementById('editor').value = e.target.result;
            document.getElementById('preview').innerHTML = e.target.result;
          };
          reader.readAsText(file);
        } else if (fileName.endsWith('.zip')) {
          JSZip.loadAsync(file).then(function(zip) {
            // Find index.html or the first .html file in the zip
            var htmlFile = zip.file("index.html") || Object.values(zip.files).find(f => f.name.endsWith('.html'));

            if (htmlFile) {
              htmlFile.async("string").then(function(content) {
                document.getElementById('editor').value = content;
                document.getElementById('preview').innerHTML = content;
              });
            } else {
              alert('No HTML file found inside the ZIP archive.');
            }
          }).catch(function(err) {
            alert('Error reading zip file: ' + err.message);
          });
        }
      }
    </script>
